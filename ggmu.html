        // Trophy Counter Animation on Scroll

        function initCounters() {

            const counters = document.querySelectorAll('.counter');

            let animated = false;


            window.addEventListener('scroll', () => {

                const trophySection = document.getElementById('trophies');

                if (!trophySection) return;

                const rect = trophySection.getBoundingClientRect();

                

                if (rect.top <= window.innerHeight && !animated) {

                    animated = true;

                    counters.forEach(counter => {

                        const target = +counter.getAttribute('data-target');

                        let count = 0;

                        const increment = target / 40;

                        

                        const updateCount = () => {

                            count += increment;

                            if (count < target) {

                                counter.innerText = Math.ceil(count);

                                setTimeout(updateCount, 40);

                            } else {

                                counter.innerText = target;

                            }

                        };

                        updateCount();

                    });

                }

            });

        }


        // Interactive Web Audio API Chant Generator

        function playChantSound(type) {

            const status = document.getElementById('sound-status');

            try {

                const AudioContext = window.AudioContext || window.webkitAudioContext;

                const ctx = new AudioContext();


                if (type === 'whistle') {

                    status.innerText = "🔊 Playing Referee Whistle...";

                    const osc = ctx.createOscillator();

                    const gain = ctx.createGain();

                    osc.type = 'sine';

                    osc.frequency.setValueAtTime(2800, ctx.currentTime);

                    gain.gain.setValueAtTime(0.3, ctx.currentTime);

                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.8);

                    osc.connect(gain);

                    gain.connect(ctx.destination);

                    osc.start();

                    osc.stop(ctx.currentTime + 0.8);

                } 

                else if (type === 'glory') {

                    status.innerText = "🎺 Playing Anthem Brass Tone...";

                    const notes = [261.63, 329.63, 392.00, 523.25]; // C E G C

                    notes.forEach((freq, idx) => {

                        const osc = ctx.createOscillator();

                        const gain = ctx.createGain();

                        osc.type = 'sawtooth';

                        osc.frequency.setValueAtTime(freq, ctx.currentTime + idx * 0.2);

                        gain.gain.setValueAtTime(0.2, ctx.currentTime + idx * 0.2);

                        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + idx * 0.2 + 0.4);

                        osc.connect(gain);

                        gain.connect(ctx.destination);

                        osc.start(ctx.currentTime + idx * 0.2);

                        osc.stop(ctx.currentTime + idx * 0.2 + 0.4);

                    });

                }

                else if (type === 'goal') {

                    status.innerText = "🎉 GOAL Siren Active!";

                    const osc = ctx.createOscillator();

                    const gain = ctx.createGain();

                    osc.type = 'sawtooth';

                    osc.frequency.setValueAtTime(400, ctx.currentTime);

                    osc.frequency.linearRampToValueAtTime(800, ctx.currentTime + 0.6);

                    gain.gain.setValueAtTime(0.4, ctx.currentTime);

                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 1.2);

                    osc.connect(gain);

                    gain.connect(ctx.destination);

                    osc.start();

                    osc.stop(ctx.currentTime + 1.2);

                }

                else if (type === 'stretford') {

                    status.innerText = "📣 Stretford End Noise Synth...";

                    const bufferSize = ctx.sampleRate * 1.5;

                    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);

                    const data = buffer.getChannelData(0);

                    for (let i = 0; i < bufferSize; i++) {

                        data[i] = Math.random() * 2 - 1;

                    }

                    const noise = ctx.createBufferSource();

                    noise.buffer = buffer;

                    const filter = ctx.createBiquadFilter();

                    filter.type = 'bandpass';

                    filter.frequency.value = 600;

                    const gain = ctx.createGain();

                    gain.gain.setValueAtTime(0.5, ctx.currentTime);

                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 1.5);

                    noise.connect(filter);

                    filter.connect(gain);

                    gain.connect(ctx.destination);

                    noise.start();

                }


                setTimeout(() => { status.innerText = ""; }, 2500);


            } catch(e) {

                status.innerText = "Audio synth not supported on browser context";

            }

        }

        function renderPoll() {

            const container = document.getElementById('poll-options');

            const total = pollVotes.reduce((acc, curr) => acc + curr.votes, 0);

            

            document.getElementById('poll-total-votes').innerText = `Total Votes: ${total.toLocaleString()}`;


            container.innerHTML = pollVotes.map(opt => {

                const pct = Math.round((opt.votes / total) * 100);

                return `

                    <div onclick="votePoll(${opt.id})" class="bg-neutral-900 border border-united-border rounded-xl p-3 cursor-pointer hover:border-united-red transition-all relative overflow-hidden group">

                        <div class="absolute top-0 bottom-0 left-0 bg-united-red/20 transition-all duration-500" style="width: ${pct}%"></div>

                        <div class="flex justify-between items-center relative z-10 font-bold text-sm">

                            <span class="text-white group-hover:text-united-gold transition-colors">${opt.name}</span>

                            <span class="text-united-gold font-mono">${pct}%</span>

                        </div>

                    </div>

                `;

            }).join('');

        }


        function votePoll(id) {

            const item = pollVotes.find(p => p.id === id);

            if (item) {

                item.votes++;

                renderPoll();

            }

        }
